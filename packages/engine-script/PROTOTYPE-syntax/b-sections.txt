# ── Declarations (one shared file) ──
Recall talked_to_campfire: bool = false
Recall was_polite_to_campfire: bool = false
Recall talked_to_campfire_trash: bool = false
Recall has_firewood: bool = false
Recall companion.Guard: Character? = none


# ── CampFire Entity ──
Dialogue CampFire
  Let met: bool = Recall.talked_to_campfire

  If Recall.talked_to_campfire_trash
    Chat
      Campfire (Angry): Go away!
    end
    Stop
  end

  Chat
    If met
      Campfire (Neutral): @campfire.greeting_returning
    Else
      Campfire (Happy): @campfire.greeting
    end
  end

  Choice
    Option "Nice to meet you!" when not met
      Recall.talked_to_campfire = true
      Recall.was_polite_to_campfire = true
    end

    Option "Sup man?" when not met
      Recall.talked_to_campfire = true
    end

    Option "Ask what the campfire has seen" when Recall.was_polite_to_campfire
      Chat
        Narrator (Sad): The fire dims for a moment, as if remembering something older than words.
        Campfire (Neutral): I remember the first Student who ever spoke to me.
      end
      Choice
        Option "Ask what happened to them" end
        Option "Never mind" end
      end
      Chat
        Campfire (Sad): ...they moved on, same as you will.
      end
    end

    Option "Ask for a story"
      Play Cutscene.campfire_story
    end

    Option "Ask Campfire to show CG again"
      If Recall.was_polite_to_campfire
        Show CG.campfire_vision
      Else
        Chat
          Campfire: The flames don't stir for someone who was rude to them.
        end
      end
    end

    Option "Warm your hands" locked unless Recall.has_firewood because "You need firewood first."
    end

    Option "What's this stupid campfire"
      Recall.talked_to_campfire_trash = true
    end
  end
end
